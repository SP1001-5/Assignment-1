--5. Create a user defined function calculate Bonus for all employees of a
--given dept using 	following conditions
create or alter function dbo.calculateBonus (@deptno int, @sal decimal)
returns decimal
as 
  begin
  declare @bonus decimal
  --a. For Deptno 10 employees 15% of sal as bonus.
  if @deptno = 10 
	set @bonus = @sal + ((@sal/100)*15)
  --b. For Deptno 20 employees  20% of sal as bonus
  else if @deptno = 20
	set @bonus = @sal + ((@sal/100)*20)
  --c For Others employees 5%of sal as bonus
  else 
	set @bonus = @sal + ((@sal/100)*5)
  return @bonus
end

select dept_id,ename,salary,dbo.calculateBonus(dept_id,salary) as bonus
from emp
where dept_id = 25
