--1.	Write a T-SQL Program to find the factorial of a given number.
declare @number int = 5
declare @counter int = 1
declare @factorial int = 1

if @number < 0
begin 
	print 'The number is less than zero which is not used to find factorial'
	end
else if @number = 0
begin
	print 'The factorial of 0 is 1'
end
else 
begin 
	while @counter <= @number
	begin
		set @factorial = @factorial * @counter
		set @counter = @counter + 1
	end
end

print 'The factorial of ' + cast(@number as varchar(20)) + ' is ' + cast(@factorial as varchar(20))

--2.	Create a stored procedure to generate multiplication table that accepts a number 
--and generates up to a given number. 
create or alter procedure MultiplicationTable @number int, @limit int
as
  begin
  declare @variable int = 1
  declare @result int 
  while @variable <= @limit
  begin 
    set @result = @number * @variable
    print cast(@number as varchar(10)) + ' * ' + cast(@variable as varchar(10)) + ' = '
           + cast(@result as varchar(10))
    set @variable = @variable + 1
   end
end

EXEC MultiplicationTable @number = 7, @limit = 10;

create table Marks (Mid int primary key,Sid int, score int,

create or alter function dbo.Getstatus (@score int)
returns varchar(10)
as
  begin
  declare @status varchar(20)
  if @score >= 50
    set @status = 'PASS'
  else
    set @status = 'FAIL'

return @status
end

select s.Sid,s.Sname, m.score,dbo.Getstatus(m.score) as Status
from Student s join Marks m on s.Sid = m.Sid
order by s.Sid
